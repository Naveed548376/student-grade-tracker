package com.mycompany.codealfa_1;

/**
 *
 * @author Naveed
 */
import java.util.ArrayList;
import java.util.InputMismatchException;
import java.util.Scanner;

/**
 * Student Grade Tracker
 * -----------------------------------------------------
 * A console-based Java program to input and manage student grades.
 * Supports adding students, entering multiple scores per student,
 * calculating average/highest/lowest scores, and printing a
 * summary report for all students.
 */
public class CodeAlfa_1 {

    // ---------- Student model ----------
    static class Student {
        String name;
        ArrayList<Double> scores = new ArrayList<>();

        Student(String name) {
            this.name = name;
        }

        void addScore(double score) {
            scores.add(score);
        }

        double getAverage() {
            if (scores.isEmpty()) return 0.0;
            double sum = 0;
            for (double s : scores) sum += s;
            return sum / scores.size();
        }

        double getHighest() {
            if (scores.isEmpty()) return 0.0;
            double max = scores.get(0);
            for (double s : scores) if (s > max) max = s;
            return max;
        }

        double getLowest() {
            if (scores.isEmpty()) return 0.0;
            double min = scores.get(0);
            for (double s : scores) if (s < min) min = s;
            return min;
        }

        String getGrade() {
            double avg = getAverage();
            if (avg >= 90) return "A";
            if (avg >= 80) return "B";
            if (avg >= 70) return "C";
            if (avg >= 60) return "D";
            return "F";
        }
    }

    // ---------- Program state ----------
    static ArrayList<Student> students = new ArrayList<>();
    static Scanner scanner = new Scanner(System.in);

    public static void main(String[] args) {
        boolean running = true;

        System.out.println("=========================================");
        System.out.println("      STUDENT GRADE TRACKER");
        System.out.println("=========================================");

        while (running) {
            printMenu();
            int choice = readInt("Enter your choice: ");

            switch (choice) {
                case 1 -> addStudent();
                case 2 -> addScoresToStudent();
                case 3 -> viewStudentReport();
                case 4 -> viewSummaryReport();
                case 5 -> {
                    System.out.println("\nExiting... Goodbye!");
                    running = false;
                }
                default -> System.out.println("Invalid choice. Please select 1-5.");
            }
        }

        scanner.close();
    }

    // ---------- Menu ----------
    static void printMenu() {
        System.out.println("\n----------------- MENU -----------------");
        System.out.println("1. Add a new student");
        System.out.println("2. Add score(s) to a student");
        System.out.println("3. View one student's report");
        System.out.println("4. View summary report (all students)");
        System.out.println("5. Exit");
        System.out.println("-----------------------------------------");
    }

    // ---------- Option 1: Add student ----------
    static void addStudent() {
        System.out.print("Enter student name: ");
        String name = scanner.nextLine().trim();

        if (name.isEmpty()) {
            System.out.println("Name cannot be empty.");
            return;
        }

        Student student = new Student(name);
        students.add(student);
        System.out.println("Student \"" + name + "\" added successfully.");

        System.out.print("Would you like to enter scores now? (y/n): ");
        String ans = scanner.nextLine().trim().toLowerCase();
        if (ans.startsWith("y")) {
            enterScores(student);
        }
    }

    // ---------- Option 2: Add scores to existing student ----------
    static void addScoresToStudent() {
        Student student = selectStudent();
        if (student == null) return;
        enterScores(student);
    }

    static void enterScores(Student student) {
        System.out.println("Enter scores for " + student.name
                + " (enter a negative number to stop):");
        while (true) {
            double score = readDouble("  Score: ");
            if (score < 0) break;
            if (score > 100) {
                System.out.println("  Score seems unusually high (>100). Enter anyway? (y/n): ");
                String ans = scanner.nextLine().trim().toLowerCase();
                if (!ans.startsWith("y")) continue;
            }
            student.addScore(score);
        }
        System.out.println("Scores recorded for " + student.name + ".");
    }

    // ---------- Option 3: View one student ----------
    static void viewStudentReport() {
        Student student = selectStudent();
        if (student == null) return;

        System.out.println("\n----- Report for " + student.name + " -----");
        if (student.scores.isEmpty()) {
            System.out.println("No scores recorded yet.");
            return;
        }
        System.out.println("Scores  : " + student.scores);
        System.out.printf("Average : %.2f%n", student.getAverage());
        System.out.printf("Highest : %.2f%n", student.getHighest());
        System.out.printf("Lowest  : %.2f%n", student.getLowest());
        System.out.println("Grade   : " + student.getGrade());
    }

    // ---------- Option 4: Summary report ----------
    static void viewSummaryReport() {
        if (students.isEmpty()) {
            System.out.println("\nNo students added yet.");
            return;
        }

        System.out.println("\n===================== SUMMARY REPORT =====================");
        System.out.printf("%-15s %-10s %-10s %-10s %-8s %-6s%n",
                "Name", "Average", "Highest", "Lowest", "#Scores", "Grade");
        System.out.println("------------------------------------------------------------");

        double classTotal = 0;
        int classCount = 0;
        double classHighest = Double.MIN_VALUE;
        double classLowest = Double.MAX_VALUE;

        for (Student s : students) {
            if (s.scores.isEmpty()) {
                System.out.printf("%-15s %-10s %-10s %-10s %-8d %-6s%n",
                        s.name, "-", "-", "-", 0, "-");
                continue;
            }
            double avg = s.getAverage();
            double high = s.getHighest();
            double low = s.getLowest();

            System.out.printf("%-15s %-10.2f %-10.2f %-10.2f %-8d %-6s%n",
                    s.name, avg, high, low, s.scores.size(), s.getGrade());

            classTotal += avg;
            classCount++;
            if (high > classHighest) classHighest = high;
            if (low < classLowest) classLowest = low;
        }

        System.out.println("------------------------------------------------------------");
        if (classCount > 0) {
            System.out.printf("Class average of averages : %.2f%n", classTotal / classCount);
            System.out.printf("Class highest score        : %.2f%n", classHighest);
            System.out.printf("Class lowest score          : %.2f%n", classLowest);
        } else {
            System.out.println("No scores have been entered for any student yet.");
        }
        System.out.println("=============================================================");
    }

    // ---------- Helpers ----------
    static Student selectStudent() {
        if (students.isEmpty()) {
            System.out.println("No students available. Please add a student first.");
            return null;
        }

        System.out.println("\nSelect a student:");
        for (int i = 0; i < students.size(); i++) {
            System.out.println("  " + (i + 1) + ". " + students.get(i).name);
        }

        int index = readInt("Enter student number: ") - 1;
        if (index < 0 || index >= students.size()) {
            System.out.println("Invalid selection.");
            return null;
        }
        return students.get(index);
    }

    static int readInt(String prompt) {
        while (true) {
            System.out.print(prompt);
            String line = scanner.nextLine().trim();
            try {
                return Integer.parseInt(line);
            } catch (NumberFormatException e) {
                System.out.println("Please enter a valid whole number.");
            }
        }
    }

    static double readDouble(String prompt) {
        while (true) {
            System.out.print(prompt);
            String line = scanner.nextLine().trim();
            try {
                return Double.parseDouble(line);
            } catch (NumberFormatException e) {
                System.out.println("Please enter a valid number.");
            }
        }
    }
}
